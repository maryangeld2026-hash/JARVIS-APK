# Jarvis 3.29 · whisper.cpp para el dictado sin internet (estático: su ggml no choca con el de llama.cpp).
cmake_minimum_required(VERSION 3.22.1)
project(jarvis_whisper C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(WHISPER_TAG "v1.9.5" CACHE STRING "versión de whisper.cpp")
set(WHISPER_DIR "" CACHE PATH "whisper.cpp ya descargado (pruebas en el computador)")

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)

if(WHISPER_DIR)
    add_subdirectory(${WHISPER_DIR} whisper-build)
else()
    include(FetchContent)
    FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG ${WHISPER_TAG} GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(whisper)
endif()

add_library(jarvis_whisper SHARED jarvis_whisper.cpp)
target_link_libraries(jarvis_whisper PRIVATE whisper)
if(ANDROID)
    target_link_libraries(jarvis_whisper PRIVATE android log)
    target_compile_options(jarvis_whisper PRIVATE -O3)
    # el ggml de whisper queda escondido dentro de esta biblioteca
    target_link_options(jarvis_whisper PRIVATE -Wl,--exclude-libs,ALL -Wl,--gc-sections)
else()
    find_package(JNI REQUIRED)
    target_include_directories(jarvis_whisper PRIVATE ${JNI_INCLUDE_DIRS})
endif()
